          payload['opId'] ??= r['op_id'];

  Future<int> deadLetterOutboxCount() async {
    final d=await db;
    final rows=await d.rawQuery('SELECT COUNT(*) AS n FROM outbox_dead_letter');
    return (rows.first['n'] as num?)?.toInt()??0;
  }
