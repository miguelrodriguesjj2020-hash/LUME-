  Future<void> update(String id) async {
    setState(()=>busy.add(id));
    try{
      await widget.services.mediaUpdates.update(id);
      if(mounted)ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content:Text('Atualização concluída.')));
    }catch(e){
      if(mounted)ScaffoldMessenger.of(context).showSnackBar(SnackBar(content:Text('Não foi possível atualizar agora: $e')));
    }finally{
      if(mounted){setState(()=>busy.remove(id));reload();}
    }
  }
